* Simple Softmax

The online softmax update rule:

#+begin_example
d_new = d_old * exp(m_old - m_new) + exp(x_new - m_new)
#+end_example

Lets see how the old term vanishes completely.

#+begin_example
e^(x - m_old) * e^(m_old - m_new)

= e^(x - 3)   * e^(3 - 4)

= e^(x - 3 + 3 - 4)

= e^(x - 4)
#+end_example

* Normal vs Online Softmax in C++

#+begin_src cpp :tangle main.cpp
#include <algorithm>
#include <cmath>
#include <iostream>
#include <vector>
#include <iomanip>


void normalSoftmax()
{
    std::cout << std::setprecision(15);
    std::vector<double> vec = {10.0f, 42.0f, 5.0f, 23.0f, 17.0f};
    auto max_it = std::max_element(vec.begin(), vec.end());
    double max_val = *max_it;

    double denom = 0.0f;

    for (float val : vec) {
        denom += std::exp(val - max_val);
    }

    std::vector<double> probability_score;
    probability_score.reserve(vec.size());

    for (double val : vec) {
        double num = std::exp(val - max_val);
        probability_score.push_back(num / denom);
    }

    for (double p : probability_score) {
        std::cout << p << '\n';
    }
}


void onlineSoftmax()
{
    std::cout << std::setprecision(15);
    std::vector<double> vec = {10.0f, 42.0f, 5.0f, 23.0f, 17.0f};
    double m = -std::numeric_limits<double>::infinity();
    double d = 0.0;

    for (double val : vec) {
        double m_new = std::max(m, val);
        //            old
        d = d * std::exp(m - m_new) + std::exp(val - m_new);
        m = m_new;
    }

    std::cout << "Softmax outputs:" << std::endl;
    for (double val : vec) {
        std::cout << std::exp(val - m) / d << std::endl;
    }
}


int main() {
  std::cout << "Normal softmax " << std::endl;
   normalSoftmax();

   std::cout << "Online softmax " << std::endl;
   onlineSoftmax();
}

* Warp Reducer

We have the warp level reducer. Now you have that local m and d from each thread.
Each thread's value of =local_d= and =local_m= is the local among the four values
that it has touched. Now from that memory location we are doing tree level reduction in warp.

#+begin_src cpp
__device__ __forceinline__ void warpReducer(float &m, float &d)
{
    for (int offset = 16; offset > 0; offset >>= 1)
    {
        float other_m = __shfl_down_sync(0xffffffff, m, offset);
        float other_d = __shfl_down_sync(0xffffffff, d, offset);

        float m_new = fmaxf(m, other_m);
        d = d * __expf(m - m_new) + other_d * __expf(other_m - m_new);
        m = m_new;
    }
}

* Full CUDA Kernel

Full simple cuda kernel. Actual production level kernel for softmax itself
might be 1000s of lines of code. This is simple and fast enough to write
an attention mechanism.

In level 1 we have a loop which touches some unique elements in memory.
From that touching we have the local max and d among those.
We do a warp level reduction and write it down to shared memory.

// ------- Kalming init -----------
class Initializer
{
public:
    std::vector<std::vector<float>> HeInit(int vocabSize, int dModel)
    {
        float stdDev = std::sqrt(2.0f / dModel);

        std::vector<std::vector<float>> tokenEmbeddings(
            vocabSize,
            std::vector<float>(dModel));

        std::random_device rd;
        std::mt19937 gen(rd());
        std::normal_distribution<float> gaussian(0.0f, 1.0f);

        for (int i = 0; i < vocabSize; ++i)
        {
            for (int j = 0; j < dModel; ++j)
            {
                tokenEmbeddings[i][j] = gaussian(gen);
            }
        }
        return tokenEmbeddings;
    }
};


// ----------- Useful utility methods -------------
class Utility {
    public:
      template <typename T>
      void Print2DVector(const std::vector<std::vector<T>> &vec, bool shape_only = false)
      {
          if (shape_only == false)
          {
              for (const auto &row : vec)
              {
                  for (const auto &value : row)
                  {
                      std::cout << value << " ";
                  }
                  std::cout << '\n';
              }
          }
          std::string shapeAnnotation = shape_only == false ? "\n] Shape: (" : "Shape: (";

          std::cout << shapeAnnotation
                    << vec.size()
                    << ", "
                    << (vec.empty() ? 0 : vec[0].size())
                    << ")\n";
      }

    template <typename T>
    T *TwoDVectorToFlatMem(const std::vector<std::vector<T>> &arr,
                           bool shape = false)
    {
        if (arr.empty())
            return nullptr;

        size_t rows = arr.size();
        size_t cols = arr[0].size();

        for (const auto &row : arr)
        {
            if (row.size() != cols)
            {
                throw std::runtime_error(
                    "TwoDVectorToFlatMem: jagged vectors are not supported.");
            }
        }

        T *newArr = new T[rows * cols];

        if (shape)
        {
            std::cout << "Shape: ("
                      << rows << ", "
                      << cols << ")\n";
        }

        for (size_t r = 0; r < rows; ++r)
        {
            for (size_t c = 0; c < cols; ++c)
            {
                newArr[r * cols + c] = arr[r][c];
            }
        }

        return newArr;
    }

    template <typename T>
    void printFlatArray2D(const T *arr, int seq_len, int d_model)
    {
        const int M = 16;

        printf("tensor([\n");
        for (int r = 0; r < seq_len; r++)
        {
            if (r == M && seq_len > M * 2)
            {
                printf("  ...,\n");
                r = seq_len - M;
            }
            printf("  [");
            for (int c = 0; c < d_model; c++)
            {
                if (c == M && d_model > M * 2)
                {
                    printf("..., ");
                    c = d_model - M;
                }
                printf("%g%s", (double)arr[r * d_model + c], c < d_model - 1 ? ", " : "");
            }
            printf("],\n");
        }
        printf("], shape=[%d, %d])\n", seq_len, d_model);
    }
};

__device__ __forceinline__ void warpReducer(float &m, float &d)
{
    for (int offset = 16; offset > 0; offset >>= 1)
    {
        float other_m = __shfl_down_sync(0xffffffff, m, offset);
        float other_d = __shfl_down_sync(0xffffffff, d, offset);

        float m_new = fmaxf(m, other_m);
        d = d * __expf(m - m_new) + other_d * __expf(other_m - m_new);
        m = m_new;
    }
}

__global__ void SoftmaxKernel2D(
    float *arr, // (B, T, vocab_size)
    float *out,
    int N,
    int seq_len,
    int vocab_size)
{
    int batch_idx = blockIdx.y;
    int row_idx = blockIdx.x;

    const float *row = arr + batch_idx * (seq_len * vocab_size) + row_idx * vocab_size;
    float *out_row = out + batch_idx * (seq_len * vocab_size) + row_idx * vocab_size;

    int lane = threadIdx.x % 32;      // position within warp
    int warp_id = threadIdx.x / 32;   // position within block
    int num_warps = blockDim.x / 32;  // total warps available

    __shared__ float smem_m[32];
    __shared__ float smem_d[32];

    float local_m = -FLT_MAX;
    float local_d = 0.0f;

    for (int i = threadIdx.x; i < vocab_size; i += blockDim.x)
    {
        float val = row[i];
        float m_new = fmaxf(local_m, val);
        local_d = local_d * __expf(local_m - m_new) + __expf(val - m_new);
        local_m = m_new;
    }

    warpReducer(local_m, local_d);

    if (lane == 0)
    {
        smem_m[warp_id] = local_m;
        smem_d[warp_id] = local_d;
    }
    __syncthreads();

    if (warp_id == 0)
    {
        local_m = (lane < num_warps) ? smem_m[lane] : -FLT_MAX;
        local_d = (lane < num_warps) ? smem_d[lane] : 0.0f;

        warpReducer(local_m, local_d);

        if (lane == 0)
        {
            smem_m[0] = local_m;
            smem_d[0] = local_d;
        }
    }
    __syncthreads();

    float global_m = smem_m[0];
    float global_d = smem_d[0];

    for (int i = threadIdx.x; i < vocab_size; i += blockDim.x)
    {
        out_row[i] = __expf(row[i] - global_m) / global_d;
    }
}


void softmaxWrapper(
  float *arr,
  float *out,
  int N,
  int seq_len,
  int vocab_size
)
{
  dim3 grid(seq_len);
  dim3 block(32);

  SoftmaxKernel2D<<<grid, block>>>(arr, out, N, seq_len, vocab_size);
  cudaDeviceSynchronize();
}


using namespace std;
int main()
{
    auto initializer = make_unique<Initializer>();
    auto utils = make_unique<Utility>();

    vector<std::vector<float>> matrix = initializer->HeInit(16, 8);

    float *flatArr = utils->TwoDVectorToFlatMem(matrix);

    cout << "BEFORE SOFTMAX " << endl;
    utils->printFlatArray2D(flatArr, 16, 8);

    float *deviceArr;
    float *deviceArrOut;

    cudaMalloc((void **)&deviceArrOut, 16 * 8 * sizeof(float));
    cudaMalloc((void **)&deviceArr, 16 * 8 * sizeof(float));
    cudaMemcpy(deviceArr, flatArr, 16 * 8 * sizeof(float), cudaMemcpyHostToDevice);

    softmaxWrapper(deviceArr, deviceArrOut, 1, 16, 8);

    cudaMemcpy(flatArr, deviceArrOut, 16 * 8 * sizeof(float), cudaMemcpyDeviceToHost);

    cout << "AFTER SOFTMAX " << endl;
    utils->printFlatArray2D(flatArr, 16, 8);

    delete[] flatArr;
    cudaFree(deviceArr);
    cudaFree(deviceArrOut);
    return 0;
}
